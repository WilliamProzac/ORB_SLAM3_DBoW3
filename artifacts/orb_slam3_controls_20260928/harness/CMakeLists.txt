cmake_minimum_required(VERSION 3.16)
project(ORB3Controls LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT IS_DIRECTORY "${U}")
 message(FATAL_ERROR "Pass -DU=<isolated source> -DVARIANT=official|delivered")
endif()
find_package(OpenCV 4 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
find_package(Threads REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization system)
set(g2o_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib CACHE PATH "" FORCE)
add_subdirectory(${U}/Thirdparty/g2o g2o)
if(VARIANT STREQUAL "official")
 add_subdirectory(${U}/Thirdparty/DBoW2 dbow)
 set(VOCAB DBoW2)
else()
 file(GLOB DBOW_SRC ${U}/Thirdparty/DBoW3/src/*.cpp ${U}/Thirdparty/DBoW3/src/*.c)
 add_library(DBoW3 SHARED ${DBOW_SRC})
 target_include_directories(DBoW3 PUBLIC ${OpenCV_INCLUDE_DIRS})
 target_compile_definitions(DBoW3 PRIVATE DBOW_DSO_EXPORTS)
 target_compile_options(DBoW3 PRIVATE -O3 -march=native)
 target_link_libraries(DBoW3 ${OpenCV_LIBS})
 set(VOCAB DBoW3)
endif()
set_target_properties(${VOCAB} PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
file(GLOB CORE ${U}/src/*.cc ${U}/src/*.cpp ${U}/src/CameraModels/*.cpp)
add_library(orb3_${VARIANT} SHARED ${CORE})
target_include_directories(orb3_${VARIANT} PUBLIC ${U} ${U}/include ${U}/include/CameraModels ${U}/Thirdparty/Sophus ${OpenCV_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIR} ${Pangolin_INCLUDE_DIRS})
target_compile_definitions(orb3_${VARIANT} PUBLIC COMPILEDWITHC11)
target_compile_options(orb3_${VARIANT} PUBLIC -O3 -march=native)
target_link_libraries(orb3_${VARIANT} PUBLIC ${VOCAB} g2o ${OpenCV_LIBS} ${Pangolin_LIBRARIES} ${Boost_LIBRARIES} Threads::Threads crypto)
add_executable(replay replay.cc)
target_link_libraries(replay orb3_${VARIANT})
