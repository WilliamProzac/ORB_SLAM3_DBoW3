cmake_minimum_required(VERSION 3.16)
project(GFReproduction LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release)
set(U ${CMAKE_CURRENT_SOURCE_DIR}/../upstream)
find_package(OpenCV 4 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
find_package(Armadillo REQUIRED)
find_package(Boost REQUIRED COMPONENTS system filesystem serialization thread)
find_package(Threads REQUIRED)
set(DBOW2_LIB_TYPE SHARED)
set(G2O_LIB_TYPE SHARED)
add_subdirectory(${U}/Thirdparty/DBoW2 dbow)
add_subdirectory(${U}/Thirdparty/g2o g2o)
set(NAMES System ChArUco Hashing Observability Tracking LocalMapping LoopClosing
 ORBextractor ORBmatcher FrameDrawer Converter MapPoint KeyFrame Map MapDrawer
 Optimizer PnPsolver Frame KeyFrameDatabase Sim3Solver Initializer Viewer)
foreach(n IN LISTS NAMES)
 list(APPEND SOURCES ${U}/src/${n}.cc)
endforeach()
foreach(mode baseline mih32 mih_ots)
 add_library(gf_${mode} SHARED ${SOURCES})
 target_include_directories(gf_${mode} PUBLIC ${U} ${U}/include ${OpenCV_INCLUDE_DIRS}
  ${EIGEN3_INCLUDE_DIR} ${Pangolin_INCLUDE_DIRS} ${ARMADILLO_INCLUDE_DIRS} /opt/ros/noetic/include)
 target_compile_options(gf_${mode} PUBLIC -O3 -march=native -include ${CMAKE_CURRENT_SOURCE_DIR}/compat.h)
 target_compile_definitions(gf_${mode} PUBLIC COMPILEDWITHC11 DISABLE_LOOP_CLOSURE)
 if(NOT mode STREQUAL baseline)
  target_compile_definitions(gf_${mode} PUBLIC LOCAL_SEARCH_USING_HASHING MAP_SIZE_TRIGGER_HASHING=2000)
 endif()
 if(mode STREQUAL mih_ots)
  target_compile_definitions(gf_${mode} PUBLIC ONLINE_TABLE_SELECTION DELAYED_MAP_MATCHING NUM_ACTIVE_HASHTABLES=8)
 endif()
 target_link_libraries(gf_${mode} PUBLIC DBoW2 g2o ${OpenCV_LIBS} ${Pangolin_LIBRARIES}
  ${ARMADILLO_LIBRARIES} ${Boost_LIBRARIES} Threads::Threads)
 add_executable(replay_${mode} replay.cc)
 target_link_libraries(replay_${mode} gf_${mode})
endforeach()
